# ========================================
# REDIS DEMO COMMANDS - SHOW CACHING WORKS
# ========================================

# ALWAYS START HERE:
cd /Users/sujithdugyala/Desktop/DSGP\ 2

# ========================================
# SCREENSHOT 1: Show Redis is Running
# ========================================
docker-compose ps redis

# ========================================
# SCREENSHOT 2: Show Number of Cached Keys
# ========================================
docker-compose exec redis redis-cli DBSIZE

# ========================================
# SCREENSHOT 3: Show Sample Cache Keys
# ========================================
docker-compose exec redis redis-cli KEYS '*' | head -20

# ========================================
# SCREENSHOT 4: Show Cache Hit/Miss Stats (PROVES PERFORMANCE!)
# ========================================
docker-compose exec redis redis-cli INFO stats | grep -E "keyspace_hits|keyspace_misses"

# ========================================
# SCREENSHOT 5: Show Memory Usage
# ========================================
docker-compose exec redis redis-cli INFO memory | grep -E "used_memory_human"

# ========================================
# SCREENSHOT 6: Show Sample Cached Data
# ========================================
# Get a sample key
docker-compose exec redis redis-cli KEYS '*' | head -1

# Then get its value (replace KEY_NAME with actual key from above)
docker-compose exec redis redis-cli GET "KEY_NAME"
